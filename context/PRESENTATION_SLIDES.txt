Database Architecture

• RDBMS: MySQL 8.4 (InnoDB Storage Engine)
• Core Entities: 16 tables, 15 ENUM types
  - User, Applicant, Bank, Branch, Account, Product
  - Application, Verification, Document, RiskAssessment
  - Offer, Contract, Disbursement
  - RepaymentSchedule, RepaymentInstallment, Payment

Design Goals:
✓ Data integrity (FK constraints, ENUMs)
✓ Auditability (timestamps, status tracking)
✓ Query efficiency (indexes, covering indexes)
✓ Transaction safety (ACID compliance)

Query Categories:
• OLTP: Real-time transactions (CRUD operations)
• Analytics: Reporting dashboards (aggregates, joins)

```sql
SELECT 
  CASE
    WHEN DATEDIFF(NOW(), due_date) BETWEEN 1 AND 30 THEN '1-30 days'
    WHEN DATEDIFF(NOW(), due_date) BETWEEN 31 AND 60 THEN '31-60 days'
    WHEN DATEDIFF(NOW(), due_date) BETWEEN 61 AND 90 THEN '61-90 days'
```

SLIDE 10: TRIGGERS & STORED ROUTINES (Logic in DB)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. TRIGGER: Auto-update application status when offer accepted

```sql
DELIMITER $$
CREATE TRIGGER trg_offer_accepted
AFTER UPDATE ON offer
FOR EACH ROW
BEGIN
  IF NEW.status = 'ACCEPTED' AND OLD.status != 'ACCEPTED' THEN
    UPDATE application
    SET status = 'OFFER_ACCEPTED', updated_at = NOW()
    WHERE application_id = NEW.application_id;
  END IF;
END$$
DELIMITER ;
```

2. TRIGGER: Mark installment as OVERDUE automatically

```sql
DELIMITER $$
CREATE TRIGGER trg_check_overdue_installments
AFTER INSERT ON repayment_installment
FOR EACH ROW
BEGIN
  -- Event scheduler checks daily:
  UPDATE repayment_installment
  SET status = 'OVERDUE'
  WHERE due_date < CURDATE() 
    AND status = 'PENDING';
END$$
DELIMITER ;
```

3. STORED PROCEDURE: Generate repayment schedule

```sql
DELIMITER $$
CREATE PROCEDURE sp_generate_repayment_schedule(
  IN p_contract_id INT
)
BEGIN
  DECLARE v_amount DECIMAL(15,2);
  DECLARE v_rate DECIMAL(5,4);
  DECLARE v_months INT;
  DECLARE v_monthly_payment DECIMAL(15,2);
  DECLARE v_schedule_id INT;
  DECLARE v_counter INT DEFAULT 1;
  DECLARE v_principal DECIMAL(15,2);
  DECLARE v_interest DECIMAL(15,2);
  DECLARE v_balance DECIMAL(15,2);
  
  -- Get contract details
  SELECT o.offered_amount, o.annual_interest_rate / 100 / 12, 
         o.offered_term_months, o.monthly_payment
  INTO v_amount, v_rate, v_months, v_monthly_payment
  FROM contract c
  INNER JOIN offer o ON c.offer_id = o.offer_id
  WHERE c.contract_id = p_contract_id;
  
  SET v_balance = v_amount;
  
  -- Create schedule header
  INSERT INTO repayment_schedule (contract_id, total_installments, 
                                   total_principal, total_interest)
  VALUES (p_contract_id, v_months, v_amount, 
          (v_monthly_payment * v_months) - v_amount);
  SET v_schedule_id = LAST_INSERT_ID();
  
  -- Generate installments (amortization)
  WHILE v_counter <= v_months DO
    SET v_interest = v_balance * v_rate;
    SET v_principal = v_monthly_payment - v_interest;
    SET v_balance = v_balance - v_principal;
    
    INSERT INTO repayment_installment (
      schedule_id, installment_number, due_date,
      principal_amount, interest_amount, total_amount, status
    )
    VALUES (
      v_schedule_id, v_counter, 
      DATE_ADD(CURDATE(), INTERVAL v_counter MONTH),
      v_principal, v_interest, v_monthly_payment, 'PENDING'
    );
    
    SET v_counter = v_counter + 1;
  END WHILE;
END$$
DELIMITER ;
```

4. FUNCTION: EMI calculator

```sql
DELIMITER $$
CREATE FUNCTION fn_calculate_emi(
  p_principal DECIMAL(15,2),
  p_annual_rate DECIMAL(5,2),
  p_months INT
)
RETURNS DECIMAL(15,2)
DETERMINISTIC
BEGIN
  DECLARE v_monthly_rate DECIMAL(10,6);
  DECLARE v_emi DECIMAL(15,2);
  
  SET v_monthly_rate = p_annual_rate / 100 / 12;
  
  IF v_monthly_rate = 0 THEN
    SET v_emi = p_principal / p_months;
  ELSE
    SET v_emi = p_principal * v_monthly_rate * 
                POWER(1 + v_monthly_rate, p_months) /
                (POWER(1 + v_monthly_rate, p_months) - 1);
  END IF;
  
  RETURN ROUND(v_emi, 2);
END$$
DELIMITER ;

-- Usage:
SELECT fn_calculate_emi(50000, 12.5, 36) AS monthly_payment;
-- Result: 1609.12
```

SLIDE 11: INTEGRITY & VALIDATION VIA QUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. ENFORCE allowed state transitions

```sql
-- Only allow SUBMITTED → VERIFICATION_IN_PROGRESS
UPDATE application
SET status = 'VERIFICATION_IN_PROGRESS', verifier_id = 3
WHERE application_id = 101
  AND status = 'SUBMITTED';  -- Constraint enforced in WHERE

-- Reject invalid transition (will affect 0 rows)
UPDATE application
SET status = 'DISBURSED'
WHERE application_id = 101
  AND status = 'SUBMITTED';  -- ❌ Invalid, no rows updated
```

2. USE ENUM for status fields (schema-level validation)

```sql
status ENUM(
  'DRAFT', 'SUBMITTED', 'UNDER_REVIEW', 
  'VERIFICATION_IN_PROGRESS', 'VERIFIED', 'REJECTED',
  'RISK_ASSESSMENT_IN_PROGRESS', 'RISK_ASSESSED',
  'OFFER_GENERATED', 'OFFER_SENT', 'OFFER_ACCEPTED', 
  'OFFER_REJECTED', 'CONTRACT_CREATED', 'CONTRACT_SIGNED',
  'DISBURSED', 'ACTIVE', 'COMPLETED', 'DEFAULTED', 
  'WRITTEN_OFF', 'CANCELLED'
) NOT NULL DEFAULT 'DRAFT'
```

3. USE TRANSACTIONS for atomic operations

```sql
START TRANSACTION;

-- Step 1: Create contract
INSERT INTO contract (offer_id, status) VALUES (201, 'SIGNED');
SET @contract_id = LAST_INSERT_ID();

-- Step 2: Generate repayment schedule
CALL sp_generate_repayment_schedule(@contract_id);

-- Step 3: Update application status
UPDATE application SET status = 'CONTRACT_SIGNED'
WHERE application_id = (
  SELECT application_id FROM offer WHERE offer_id = 201
);

-- Step 4: Log event
INSERT INTO audit_log (entity_type, entity_id, action, user_id)
VALUES ('CONTRACT', @contract_id, 'CREATED', 2);

COMMIT;
```

4. CHECK constraints (MySQL 8.0.16+)

```sql
ALTER TABLE offer
ADD CONSTRAINT chk_offered_amount_positive
CHECK (offered_amount > 0);

ALTER TABLE repayment_installment
ADD CONSTRAINT chk_installment_amounts
CHECK (total_amount = principal_amount + interest_amount);
```

SLIDE 12: INDEXING & PERFORMANCE QUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. DEFINE indexes for frequent joins

```sql
-- Accelerate applicant → applications lookup
CREATE INDEX idx_application_applicant 
ON application(applicant_id);

-- Accelerate application → documents lookup
CREATE INDEX idx_document_application 
ON document(application_id);

-- Accelerate contract → repayment schedule
CREATE INDEX idx_schedule_contract 
ON repayment_schedule(contract_id);

-- Composite index for status filtering + sorting
CREATE INDEX idx_application_status_created 
ON application(status, created_at DESC);
```

2. COVERING index for dashboard queries

```sql
-- Index covers SELECT without table access
CREATE INDEX idx_application_dashboard
ON application(status, product_id, created_at, requested_amount);

-- Query uses covering index:
SELECT status, COUNT(*), SUM(requested_amount)
FROM application
WHERE status IN ('SUBMITTED', 'VERIFICATION_IN_PROGRESS')
GROUP BY status;
```

```sql
EXPLAIN SELECT
```

```sql
-- Slow: Full table scan

-- Fast: Index-only scan

SELECT COUNT(*) FROM application
```

```sql
CREATE VIEW v_application_full AS
```

2. VIEW: Active portfolio summary

```sql
CREATE VIEW v_active_portfolio AS
SELECT 
  c.contract_id,
  ap.full_name AS borrower,
  o.offered_amount AS principal,
  o.annual_interest_rate,
  rs.total_installments,
  COUNT(ri.installment_id) AS paid_installments,
  SUM(CASE WHEN ri.status = 'PAID' THEN ri.total_amount ELSE 0 END) AS paid_amount,
  SUM(CASE WHEN ri.status = 'PENDING' THEN ri.total_amount ELSE 0 END) AS outstanding
FROM contract c
INNER JOIN offer o ON c.offer_id = o.offer_id
INNER JOIN application a ON o.application_id = a.application_id
INNER JOIN applicant ap ON a.applicant_id = ap.applicant_id
INNER JOIN repayment_schedule rs ON c.contract_id = rs.contract_id
LEFT JOIN repayment_installment ri ON rs.schedule_id = ri.schedule_id
WHERE c.status = 'ACTIVE'
GROUP BY c.contract_id, ap.full_name, o.offered_amount, 
         o.annual_interest_rate, rs.total_installments;

-- Usage:
SELECT * FROM v_active_portfolio WHERE outstanding > 10000;
```

3. VIEW: Overdue installments (risk monitoring)

```sql
CREATE VIEW v_overdue_installments AS
SELECT 
  ri.installment_id,
  ri.due_date,
  DATEDIFF(CURDATE(), ri.due_date) AS days_overdue,
  ri.total_amount AS overdue_amount,
  ap.full_name AS borrower,
  ap.phone AS contact_phone,
  c.contract_id
FROM repayment_installment ri
INNER JOIN repayment_schedule rs ON ri.schedule_id = rs.schedule_id
INNER JOIN contract c ON rs.contract_id = c.contract_id
INNER JOIN offer o ON c.offer_id = o.offer_id
INNER JOIN application a ON o.application_id = a.application_id
INNER JOIN applicant ap ON a.applicant_id = ap.applicant_id
WHERE ri.status = 'OVERDUE'
ORDER BY days_overdue DESC;

-- Usage:
SELECT * FROM v_overdue_installments WHERE days_overdue > 30;
```

SLIDE 15: ADVANCED QUERIES (Window Functions & CTEs)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. WINDOW FUNCTION: Rank applications by amount within each product

```sql
SELECT 
  application_id,
  product_id,
  requested_amount,
  RANK() OVER (
    PARTITION BY product_id 
    ORDER BY requested_amount DESC
  ) AS amount_rank
FROM application
WHERE status != 'DRAFT';
```

2. WINDOW FUNCTION: Running total of disbursements

```sql
SELECT 
  disbursement_id,
  contract_id,
  disbursed_amount,
  disbursed_at,
  SUM(disbursed_amount) OVER (
    ORDER BY disbursed_at
  ) AS cumulative_disbursed
FROM disbursement
WHERE status = 'COMPLETED'
ORDER BY disbursed_at;
```

3. CTE: Multi-step calculation for portfolio risk

```sql
WITH risk_summary AS (
  SELECT 
    ra.overall_risk_level,
    COUNT(*) AS loan_count,
    SUM(o.offered_amount) AS total_exposure
  FROM risk_assessment ra
  INNER JOIN offer o ON ra.application_id = o.application_id
  INNER JOIN contract c ON o.offer_id = c.offer_id
  WHERE c.status = 'ACTIVE'
  GROUP BY ra.overall_risk_level
),
total_portfolio AS (
  SELECT SUM(total_exposure) AS grand_total
  FROM risk_summary
)
SELECT 
  rs.overall_risk_level,
  rs.loan_count,
  rs.total_exposure,
  ROUND(rs.total_exposure / tp.grand_total * 100, 2) AS pct_of_portfolio
FROM risk_summary rs
CROSS JOIN total_portfolio tp
ORDER BY FIELD(rs.overall_risk_level, 'LOW', 'MEDIUM', 'HIGH', 'VERY_HIGH');
```

SLIDE 16: TRANSACTION ISOLATION & CONCURRENCY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Handling Concurrent Operations

1. SET TRANSACTION ISOLATION LEVEL

```sql
-- For critical financial operations
SET TRANSACTION ISOLATION LEVEL SERIALIZABLE;
START TRANSACTION;

-- Verify offer is still available
SELECT status, offer_valid_until 
FROM offer 
WHERE offer_id = 201 
FOR UPDATE;  -- Lock row

-- Accept offer if valid
UPDATE offer SET status = 'ACCEPTED' WHERE offer_id = 201;

COMMIT;
```

2. PESSIMISTIC LOCKING (SELECT ... FOR UPDATE)

```sql
START TRANSACTION;

-- Lock contract row to prevent concurrent modifications
SELECT * FROM contract 
WHERE contract_id = 301 
FOR UPDATE;

-- Make payment without race condition
INSERT INTO payment (installment_id, amount_paid, paid_at)
VALUES (501, 1609.12, NOW());

UPDATE repayment_installment 
SET status = 'PAID' 
WHERE installment_id = 501;

COMMIT;
```

3. OPTIMISTIC LOCKING (Version Number)

```sql
-- Add version column to application
ALTER TABLE application ADD COLUMN version INT DEFAULT 0;

-- Update with version check
UPDATE application
SET status = 'VERIFIED', 
    verifier_id = 3,
    version = version + 1
WHERE application_id = 101
  AND version = 5;  -- Only succeeds if version matches

-- Check affected rows; retry if 0
```

SLIDE 17: SECURITY & ACCESS CONTROL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Role-Based Database Users

```sql
-- 1. Create database users
CREATE USER 'applicant_user'@'%' IDENTIFIED BY 'secure_password';
CREATE USER 'banker_user'@'%' IDENTIFIED BY 'secure_password';
CREATE USER 'admin_user'@'%' IDENTIFIED BY 'secure_password';

-- 2. Grant permissions by role

-- APPLICANT: Read own data, insert applications
GRANT SELECT ON olavs.user TO 'applicant_user'@'%';
GRANT SELECT ON olavs.applicant TO 'applicant_user'@'%';
GRANT SELECT, INSERT ON olavs.application TO 'applicant_user'@'%';
GRANT SELECT ON olavs.document TO 'applicant_user'@'%';
GRANT SELECT ON olavs.offer TO 'applicant_user'@'%';
GRANT SELECT ON olavs.repayment_installment TO 'applicant_user'@'%';

-- BANKER: Full access to applications, offers, contracts
GRANT SELECT, INSERT, UPDATE ON olavs.application TO 'banker_user'@'%';
GRANT SELECT, INSERT, UPDATE ON olavs.verification TO 'banker_user'@'%';
GRANT SELECT, INSERT, UPDATE ON olavs.offer TO 'banker_user'@'%';
GRANT SELECT, INSERT, UPDATE ON olavs.contract TO 'banker_user'@'%';
GRANT SELECT ON olavs.risk_assessment TO 'banker_user'@'%';

-- ADMIN: Full access to all tables
GRANT ALL PRIVILEGES ON olavs.* TO 'admin_user'@'%';

-- 3. Revoke dangerous permissions
REVOKE DELETE ON olavs.payment FROM 'banker_user'@'%';
REVOKE DROP ON olavs.* FROM 'banker_user'@'%';
```

2. TABLE SIZE MONITORING

```sql
SELECT 
  table_name,
  ROUND(data_length / 1024 / 1024, 2) AS data_mb,
  ROUND(index_length / 1024 / 1024, 2) AS index_mb,
  ROUND((data_length + index_length) / 1024 / 1024, 2) AS total_mb,
  table_rows
FROM information_schema.tables
WHERE table_schema = 'olavs'
ORDER BY (data_length + index_length) DESC;
```
